2️⃣ Passo 2: Criar o Componente de Mídia Inspirado no Hevy/Lyfta
No Replit/GitHub, crie o arquivo src/components/workout/ExerciseMedia.tsx:

React

'use client'

import { useState } from 'react'
import { Dumbbell, Activity } from 'lucide-react'

interface Props {
  videoUrl?: string | null
  muscleGroup: string
  exerciseName: string
}

export function ExerciseMedia({ videoUrl, muscleGroup, exerciseName }: Props) {
  const [imageError, setImageError] = useState(false)

  return (
    <div className="w-full bg-zinc-900/90 border border-zinc-800 rounded-2xl overflow-hidden relative group">
      {/* Se tiver URL de vídeo/GIF válida e não der erro */}
      {videoUrl && !imageError ? (
        <div className="relative w-full h-48 bg-black flex items-center justify-center overflow-hidden">
          <img
            src={videoUrl}
            alt={exerciseName}
            onError={() => setImageError(true)}
            className="w-full h-full object-cover opacity-90 transition-scale duration-300 group-hover:scale-105"
          />
          <div className="absolute inset-0 bg-gradient-to-t from-zinc-950 via-transparent to-transparent opacity-80" />
        </div>
      ) : (
        /* Visualizador de Anatomia Muscular Minimalista (Estilo Hevy) */
        <div className="w-full h-36 bg-gradient-to-b from-zinc-900 to-zinc-950 flex flex-col items-center justify-center p-4 relative overflow-hidden">
          {/* Efeito de iluminação de fundo */}
          <div className="absolute w-32 h-32 bg-white/5 rounded-full blur-2xl top-1/2 left-1/2 -translate-x-1/2 -translate-y-1/2" />

          <div className="z-10 flex flex-col items-center gap-2">
            <div className="w-12 h-12 bg-black border border-zinc-800 rounded-2xl flex items-center justify-center shadow-2xl">
              <Dumbbell className="w-6 h-6 text-white animate-pulse" />
            </div>
            
            <div className="text-center">
              <span className="text-[10px] font-black uppercase tracking-widest text-emerald-400 bg-emerald-950/60 border border-emerald-800/40 px-3 py-1 rounded-full flex items-center gap-1.5 mx-auto">
                <Activity className="w-3 h-3" /> Foco: {muscleGroup}
              </span>
            </div>
          </div>
        </div>
      )}
    </div>
  )
}
